\section*{Vector Spaces (Ch1)}

\subsection*{Vector space}
$V$ over $\F$: $+:V\times V\to V$, $\cdot:\F\times V\to V$ s.t.
\[
\begin{aligned}
&u+v=v+u,\ (u+v)+w=u+(v+w),\ \exists 0:\ v+0=v,\ \forall v\,\exists(-v):v+(-v)=0,\\
&1v=v,\ (ab)v=a(bv),\ a(u+v)=au+av,\ (a+b)v=av+bv.
\end{aligned}
\]
Derived: $-(-u)=u,\ 0_{\F}u=0,\ (-1)u=-u,\ c\,0=0,\ u-v:=u+(-v)$.

\subsection*{Subspaces \& affine/translate criteria}
$W\le V \iff (0\in W)\ \wedge\ (u,v\in W\Rightarrow u+v\in W)\ \wedge\ (\lambda\in\F,u\in W\Rightarrow \lambda u\in W)$.
One-line: $W\le V\iff W\neq\varnothing\ \wedge\ (\forall u,v\in W,\forall c\in\F:\ cu+v\in W)$.
Consequences ($W\le V$): $0\in W,\ u\in W\Rightarrow -u\in W,\ u,v\in W\Rightarrow u-v\in W$.\\
Fast disproof: $0\notin W\Rightarrow W\not\le V$.

\textbf{Kernel trick:} $T:V\to U$ linear $\Rightarrow \Ker(T)\le V$.\\
\textbf{Affine span:} $\{ \sum_{i=1}^n\lambda_i a_i+b:\lambda_i\in\F\}\le V\iff b\in\spann\{a_1,\dots,a_n\}$.\\
\textbf{Union:} $W_1,W_2\le V$: $W_1\cup W_2\le V\iff (W_1\subseteq W_2)\ \vee\ (W_2\subseteq W_1)$.

\subsection*{Span / sum / intersection}
$\spann(S)=$ all \emph{finite} linear combinations of vectors in $S$; $\spann(\varnothing)=\{0\}$.\\
$\spann(S)\le V$; if $S\subseteq W\le V$ then $\spann(S)\subseteq W$ (smallest-subspace property).\\
For $A,B\subseteq V$:
\[
A\subseteq B\Rightarrow \spann(A)\subseteq\spann(B),\qquad
\spann(A\cup B)=\spann(A)+\spann(B).
\]
If $W_1,W_2\le V$: $W_1\cap W_2\le V$, $W_1+W_2\le V$, and $W_1+W_2=\spann(W_1\cup W_2)$.

\subsection*{Direct sums}
$W=V_1\oplus\cdots\oplus V_n$ means $W=\sum_i V_i$ and decomposition $w=\sum_i v_i$ ($v_i\in V_i$) is unique.\\
Zero-sum test:
\[
\sum_{i=1}^n v_i=0,\ v_i\in V_i\ \Rightarrow\ v_1=\cdots=v_n=0.
\]
Intersection test:
\[
V_i\cap\Bigl(\sum_{j\neq i}V_j\Bigr)=\{0\}\ \ \forall i.
\]
Two-subspace special case:
\[
W=V_1\oplus V_2 \iff (W=V_1+V_2)\ \wedge\ (V_1\cap V_2=\{0\}).
\]
Sum map: $T:V_1\times\cdots\times V_n\to V$, $T(v_1,\dots,v_n)=\sum_i v_i$,
\[
\im(T)=\sum_i V_i,\qquad \Ker(T)=\{(v_i):\sum_i v_i=0\},\qquad \sum_i 
\]
\[
V_i\text{ direct}\iff \Ker(T)=\{0\}.
\]
Three-in-one: if $S$ LI and $S=A\sqcup B$, then $\spann(S)=\spann(A)\oplus\spann(B)$.

\section*{Finite-Dimensional Spaces (Ch2)}

\subsection*{Linear independence / dependence}
$S$ LI $\iff$ for distinct $v_1,\dots,v_k\in S$:
\[
\lambda_1v_1+\cdots+\lambda_kv_k=0\Rightarrow \lambda_1=\cdots=\lambda_k=0.
\]
Dependence $\iff \exists v\in S$ with $v\in\spann(S\setminus\{v\})$ (allow $v=0$).\\
Facts: $\varnothing$ LI; $\{v\}$ LI $\iff v\neq 0$; $\{u,v\}$ dependent $\iff u=cv$.\\
Monotonicity: LI $\Rightarrow$ all subsets LI; dependent $\Rightarrow$ all supersets dependent.

\textbf{Column test in $\F^m$:} with $A=[v_1\ \cdots\ v_n]$,
\[
\{v_i\}\text{ LI}\iff A\lambda=0 \text{ has only trivial soln}\iff \operatorname{rref}(A)\ \text{pivot in every column}.
\]
Sanity: in $\F^m$, any set of $>m$ vectors is dependent.

\subsection*{Bases, coordinates, dimension}
$B$ basis $\iff$ (LI) $\wedge$ $\spann(B)=V$. Unique expansion:
\[
v=\sum_{b\in B}\lambda_b b\quad(\lambda_b=0\text{ for all but finitely many }b).
\]
Ordered basis $\mathcal{B}=(b_1,\dots,b_n)$: $v=\sum_{i=1}^n\lambda_i b_i\Rightarrow [v]_{\mathcal{B}}=(\lambda_1,\dots,\lambda_n)^{\mathsf T}$.\\
Ordering matters: if $\mathcal{B}'$ swaps $b_1,b_2$, then $[3b_1+5b_2]_{\mathcal{B}}=\binom{3}{5}$, $[3b_1+5b_2]_{\mathcal{B}'}=\binom{5}{3}$.

Basis existence; for $\dim(V)=n$:
\[
\text{LI set size}\le n,\qquad \text{spanning set size}\ge n,\qquad
\]
\[
(|S|=n)\Rightarrow (S\text{ basis}\iff S\text{ spans}\iff S\text{ LI}).
\]
Add-one-more: $S$ LI $\Rightarrow S\cup\{v\}$ LI $\iff v\notin\spann(S)$.\\
Hereditary: $W\le V$, $\dim(V)<\infty\Rightarrow \dim(W)<\infty$.\\
Basis extension: LI set extends to a basis. Spanning set contains a basis.

\textbf{Build a basis:} (extend) add vectors outside current span; (extract) remove redundant spanning vectors.\\

\subsection*{Dimension formulas \& common dimensions}
If $W\le V$ finite-dim: $\dim(W)\le \dim(V)$; equality $\iff W=V$.\\
Sum formula (finite-dim): for $U,W\le V$,
\[
\dim(U+W)=\dim(U)+\dim(W)-\dim(U\cap W),
\]
\[
\dim(U\cap W)\ge \dim(U)+\dim(W)-\dim(V).
\]
Direct sum: $V=U\oplus W \Rightarrow \dim(V)=\dim(U)+\dim(W)$, and $V=U\oplus W$ iff any two of
\[
V=U+W,\quad U\cap W=\{0\},\quad \dim(U)+\dim(W)=\dim(V).
\]

Standard dims:
\[
\dim(\F^n)=n,\quad \dim(\F^{m\times n})=mn,\quad \dim(P_n(\F))=n+1,\quad \dim(P(\F))=\infty.
\]
Matrix subspaces (over indicated fields):
\begin{align*}
& \dim\{M\in\F^{n\times n}:\tr(M)=0\}=n^2-1,\\
& \dim\{M\in\R^{n\times n}:M^{\mathsf T}=M\}=\tfrac{n(n+1)}{2},\\
& \dim\{M\in\R^{n\times n}:M^{\mathsf T}=-M\}=\tfrac{n(n-1)}{2},\\
& \dim\{M\in\R^{n\times n}:M\text{ upper tri}\}=\tfrac{n(n+1)}{2}.
\end{align*}

\subsection*{Codimension-1 (kernel of nonzero functional)}
If $L:W\to\F$ nonzero linear, then $\Ker(L)\le W$ and (finite-dim)
\[
\dim(\Ker(L))=\dim(W)-1,\qquad
\forall p\in W\ (L(p)\neq 0):\ W=\Ker(L)\oplus\spann\{p\},
\]
\[
w=\Bigl(w-\frac{L(w)}{L(p)}p\Bigr)+\frac{L(w)}{L(p)}p,\quad
w-\frac{L(w)}{L(p)}p\in\Ker(L).
\]

\section*{Linear Transformations (Ch3)}

\subsection*{Linearity \& basic consequences}
$T:V\to W$ linear iff $T(u+v)=T(u)+T(v)$ and $T(\lambda u)=\lambda T(u)$; equivalently
\[
T(\alpha u+\beta v)=\alpha T(u)+\beta T(v)\quad(\forall u,v,\alpha,\beta).
\]
Then $T(0)=0$, $T(-v)=-T(v)$, and for finite sums:
\[
T\!\left(\sum_{i=1}^k\lambda_i v_i\right)=\sum_{i=1}^k\lambda_iT(v_i).
\]
$\mathcal{L}(V,W)$ is a vector space (pointwise).\\

\subsection*{Kernel/range; rank--nullity; injective/surjective}
\[
\Ker(T)=\{v:T(v)=0\}\le V,\qquad \Range(T)=\{T(v):v\in V\}\le W.
\]
If $\dim(V)<\infty$:
\[
\dim(V)=\rank(T)+\nullity(T),\quad 
\]
\[
\rank(T)=\dim(\Range(T)),\quad \nullity(T)=\dim(\Ker(T)).
\]
Injective tests:
\[
T\text{ injective}\iff \Ker(T)=\{0\}\iff
\]
\[
\nullity(T)=0\iff \rank(T)=\dim(V).
\]
Surjective:
\[
T\text{ onto }W\iff \Range(T)=W\iff \rank(T)=\dim(W).
\]
If $\dim(V)=\dim(W)$: injective $\iff$ surjective $\iff$ bijective $\iff$ invertible.

Composition facts: $T\circ S$ injective $\Rightarrow S$ injective; $T\circ S$ surjective $\Rightarrow T$ surjective; if $S,T$ invertible then $(T\circ S)^{-1}=S^{-1}\circ T^{-1}$.\\

\subsection*{Matrix representations \& change of coordinates}
Finite-dim: ordered bases $\beta=(b_1,\dots,b_n)$ of $V$, $\gamma=(c_1,\dots,c_m)$ of $W$:
\[
[T]^{\gamma}_{\beta}=\bigl[\ [T(b_1)]_{\gamma}\ \cdots\ [T(b_n)]_{\gamma}\ \bigr]\in\F^{m\times n},
\qquad
[T(v)]_{\gamma}=[T]^{\gamma}_{\beta}[v]_{\beta}.
\]
Composition: $[T\circ S]^{\gamma}_{\alpha}=[T]^{\gamma}_{\beta}[S]^{\beta}_{\alpha}$.\\
Change of coordinates: $[v]_{\gamma}=[I_V]^{\gamma}_{\beta}[v]_{\beta}$ and $[I_V]^{\beta}_{\gamma}=([I_V]^{\gamma}_{\beta})^{-1}$.\\
Change of representation:
\[
[T]^{\gamma}_{\gamma}=[I_V]^{\gamma}_{\beta}\,[T]^{\beta}_{\beta}\,[I_V]^{\beta}_{\gamma}.
\]
Inverse: if $T$ invertible then $[T^{-1}]^{\beta}_{\gamma}=([T]^{\gamma}_{\beta})^{-1}$.\\
Compute $[I_V]^{\gamma}_{\beta}$ via standard basis $\sigma$:
\[
P_{\beta}=[I_V]^{\sigma}_{\beta}=\bigl[\ [b_1]_{\sigma}\ \cdots\ [b_n]_{\sigma}\ \bigr],\quad
[I_V]^{\gamma}_{\beta}=P_{\gamma}^{-1}P_{\beta}\quad(\text{row-reduce }[P_{\gamma}\mid P_{\beta}]).
\]
RREF criteria for $A=[T]^{\gamma}_{\beta}\in\F^{n\times m}$:
\[
T\text{ injective}\iff \operatorname{rref}(A)\text{ pivot in every column},
\]
\[
T\text{ onto }W\iff \operatorname{rref}(A)\text{ pivot in every row}.
\]
\textbf{Matrix-rep iso:} $\Phi_{\beta,\alpha}:\mathcal{L}(U,V)\to \F^{\dim(V)\times\dim(U)}$, $\Phi(S)=[S]^{\beta}_{\alpha}$ is a vector-space isomorphism.

\subsection*{Similarity, determinant, trace}
Similarity: $A\sim B\iff \exists Q\in GL_n(\F):\ B=Q^{-1}AQ$ (equivalence relation).\\
Invariants: $A\sim B\Rightarrow \det(A)=\det(B)$ and $\tr(A)=\tr(B)$, using
\[
\det(Q^{-1}AQ)=\det(A),\qquad \tr(Q^{-1}AQ)=\tr(A),\qquad \tr(XY)=\tr(YX).
\]
