\section*{Question 2 (Utility maximisation)}
\subsection*{Problem}
Suppose the consumer has utility function
\[
U(x_1,x_2)=x_1x_2.
\]
Let prices be $p=(p_1,p_2)$ and income (expenditure) be $m$, so the budget constraint is
\[
p_1x_1+p_2x_2=m,
\qquad x_1\ge 0,\ x_2\ge 0.
\]

\begin{enumerate}[label=(\alph*)]
  \item Compute the marginal utilities $MU_1$ and $MU_2$.
  \item Compute the marginal rate of substitution $MRS_{12}$.
  \item When $(p_1,p_2)=(2,1)$ and $m=1$, compute the optimal demand $(x_1,x_2)$.
  \item Derive the Marshallian demand function $x(p_1,p_2,m)$.
\end{enumerate}
